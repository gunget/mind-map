    function parseSubpageItem(p) {
        if (!p) return null;
        let name = p.name || "무제 시트";
        if (p.data) {

            try { decodedStr = decodeSubpageContent(p.data); } catch(e) { decodedStr = p.data; }

                try {
                    const obj = JSON.parse(decodedStr);
                    if (obj.nodes || obj.edges) {
                        return {
                            name: obj.name || name,
                            nodes: obj.nodes || [],
                            edges: obj.edges || [],
                            bgSettings: obj.bgSettings || { color: "#fdfbf0", showGrid: true, modalZoom: 1.0 },
                            projectId: obj.projectId || ("id_" + Date.now())
                        };
                    }
                } catch(e) {}
            }
            if (typeof decodedStr === 'string' && (decodedStr.includes("/*DATA_HOLDER*/") || decodedStr.includes("<html"))) {
                const extracted = extractDataBlock(decodedStr);

                    return {
                        name: name,

                        bgSettings: extracted.bgSettings || { color: "#fdfbf0", showGrid: true, modalZoom: 1.0 },
                        projectId: extracted.projectId || ("id_" + Date.now())
                    };
                }
            }
        }
        if (p.htmlContent) {
            const extracted = extractDataBlock(p.htmlContent);
            if (extracted) {
                return {
                    name: name,
                    nodes: extracted.nodes || [],
                    edges: extracted.edges || [],
                    bgSettings: extracted.bgSettings || { color: "#fdfbf0", showGrid: true, modalZoom: 1.0 },
                    projectId: extracted.projectId || ("id_" + Date.now())

                try { return JSON.parse(val.replace(/;$/, '')); } catch(e) { return null; }
            };
            
            state.nodes = parseVar('nodes', 'edges') || [];
            state.edges = parseVar('edges', 'bgSettings') || [];
            state.bgSettings = parseVar('bgSettings', 'subPages') || parseVar('bgSettings', 'safeSubPages') || { color: "#fdfbf0", showGrid: true, modalZoom: 1.0 };
            
            const sMatch = inner.match(/let safeSubPages = (\[[\s\S]*?\]);/);
            if (sMatch) {
                try {
                    const parsed = JSON.parse(sMatch[1]);
                    state.subPages = parsed.map(p => parseSubpageItem(p)).filter(Boolean);
                } catch(e) { console.error("서브페이지 파싱 에러(safeSubPages)", e); }
            } else {
                const oldMatch = inner.match(/let subPages = (\[[\s\S]*?\]);/);
                if (oldMatch) {
                    try {
                        const parsed = JSON.parse(oldMatch[1]);
                        state.subPages = parsed.map(p => parseSubpageItem(p)).filter(Boolean);
                    } catch(e) {}
                }
            }
        } catch(e) { console.error("Data Extraction Error:", e); }
        
        return state;
    }

    function captureState() { 
        nodes.forEach(n => { if (!n.id) n.id = generateId(); }); 
        const nodeSnapshot = JSON.parse(JSON.stringify(nodes, (k, v) => k.startsWith('_') ? undefined : v)); 
        const edgeSnapshot = edges.map(e => { 
            return { ...e, fromId: e.from ? e.from.id : null, toId: e.to ? e.to.id : null, from: null, to: null }; 
        }).filter(e => e.fromId && e.toId); 
        return { nodes: nodeSnapshot, edges: edgeSnapshot }; 
    }
    
    function applyState(state) { 
        if (!state) return; 
        nodes = state.nodes; 
        nodes.forEach(n => { if (!n.id) n.id = generateId(); }); 

        edges = state.edges.map(e => { 
            let fNode = null, tNode = null; 
            if (e.fromId && e.toId) { 
                fNode = nodes.find(n => n.id === e.fromId);
                tNode = nodes.find(n => n.id === e.toId);
            } else if (e.fromIdx !== undefined && e.fromIdx !== -1) { 
                fNode = nodes[e.fromIdx]; 
                tNode = nodes[e.toIdx]; 
            } else if (e.from && typeof e.from === 'object') { 
                fNode = nodes.find(n => Math.abs(n.x - e.from.x) < 0.1 && Math.abs(n.y - e.from.y) < 0.1); 
                tNode = nodes.find(n => Math.abs(n.x - e.to.x) < 0.1 && Math.abs(n.y - e.to.y) < 0.1); 
            } 
            return { ...e, from: fNode, to: tNode }; 
        }).filter(e => e.from && e.to); 
        selectedNodes = []; selectedEdge = null; 
        requestRender();
    }

    function performPaste(clipData) {
        if (!clipData || !clipData.nodes) return;
        saveHistory();
        const nNodes = clipData.nodes.map(n => { 
            let nn = JSON.parse(JSON.stringify(n, (k, v) => k.startsWith('_') ? undefined : v)); 
            nn.id = generateId(); 
            nn.x = currentWorldMouse.x + (n.x - clipData.cx); 
            nn.y = currentWorldMouse.y + (n.y - clipData.cy); 
            return nn; 
        });
        const nEdges = clipData.edges.map(e => { 
            let ne = JSON.parse(JSON.stringify(e)); 
            ne.from = nNodes[e.fi]; 
            ne.to = nNodes[e.ti]; 
            delete ne.fi; delete ne.ti; 
            return ne; 
        });
        nodes.push(...nNodes); edges.push(...nEdges); selectedNodes = [...nNodes];
        requestRender();
        showToast("블럭을 붙여넣었습니다.");
    }

    function copySelected() {
        if (selectedNodes.length === 0) return;
        const cNodes = JSON.parse(JSON.stringify(selectedNodes, (k, v) => k.startsWith('_') ? undefined : v)), cEdges = [];
        edges.forEach(e => { const fi = selectedNodes.indexOf(e.from), ti = selectedNodes.indexOf(e.to); if (fi !== -1 && ti !== -1) { let ne = JSON.parse(JSON.stringify(e)); ne.fi = fi; ne.ti = ti; cEdges.push(ne); } });
        let ax = 0, ay = 0; cNodes.forEach(n => { ax += n.x; ay += n.y; });
        
        const clipData = { ideaBlockCopy: true, nodes: cNodes, edges: cEdges, cx: ax/cNodes.length, cy: ay/cNodes.length };
        copyBuffer = clipData; 
        
        try {
            const textArea = document.createElement("textarea");
            textArea.value = JSON.stringify(clipData);
            document.body.appendChild(textArea);
            textArea.select();
            document.execCommand("copy");
            document.body.removeChild(textArea);
            showToast("블럭이 복사되었습니다. 다른 탭에서 Ctrl+V를 누르세요.");
        } catch (e) { showToast("동일한 창 내에서만 복사되었습니다."); }
    }

    function parseRichText(text) {
        let result = []; let bold = false, italic = false, underline = false, color = null; let i = 0;
        while (i < text.length) {
            if (text.startsWith('**', i)) { bold = !bold; i += 2; continue; } if (text.startsWith('__', i)) { underline = !underline; i += 2; continue; } if (text.startsWith('*', i)) { italic = !italic; i += 1; continue; }
            if (text.startsWith('[c:', i)) { let endIdx = text.indexOf(']', i); if (endIdx !== -1) { color = text.substring(i + 3, endIdx); i = endIdx + 1; continue; } }
            if (text.startsWith('[/c]', i)) { color = null; i += 4; continue; } result.push({ char: text[i], bold, italic, underline, color }); i++;
        } return result;
    }

    function getNodeLayout(context, node) {
        if (["marker", "image", "area"].includes(node.type)) return null;
        if (node._layoutCache &&
            node._layoutCache.text === node.text &&
            node._layoutCache.fontSize === node.fontSize &&
            node._layoutCache.width === node.width &&
            node._layoutCache.textAlign === (node.textAlign || "center")) {
            return node._layoutCache;
        }

        const padding = 28; const maxWidth = Math.max(20, node.width - padding); const styledChars = parseRichText(node.text || "");
        const lines = []; let currentLine = []; let currentWidth = 0;
        for (let i = 0; i < styledChars.length; i++) {
            let sc = styledChars[i];
            if (sc.char === '\n') { lines.push({ chars: currentLine, width: currentWidth }); currentLine = []; currentWidth = 0; continue; }
            let fontStyle = (sc.bold ? "bold " : "") + (sc.italic ? "italic " : ""); context.font = `${fontStyle}${node.fontSize}px 'Malgun Gothic'`; let w = context.measureText(sc.char).width;
            if (currentWidth + w > maxWidth && currentLine.length > 0) { lines.push({ chars: currentLine, width: currentWidth }); currentLine = [sc]; currentWidth = w; } else { currentLine.push(sc); currentWidth += w; }
        } lines.push({ chars: currentLine, width: currentWidth });

        const lh = node.fontSize * 1.5; const totalTextHeight = lines.length * lh; node.height = Math.max(60, totalTextHeight + padding);

        const preparedLines = lines.map(lineInfo => {
            let chunks = []; let currentChunk = null;
            lineInfo.chars.forEach(sc => {
                if (!currentChunk) { currentChunk = { text: sc.char, bold: sc.bold, italic: sc.italic, underline: sc.underline, color: sc.color }; }
                else if (currentChunk.bold === sc.bold && currentChunk.italic === sc.italic && currentChunk.underline === sc.underline && currentChunk.color === sc.color) { currentChunk.text += sc.char; }
                else { chunks.push(currentChunk); currentChunk = { text: sc.char, bold: sc.bold, italic: sc.italic, underline: sc.underline, color: sc.color }; }
            });
            if (currentChunk) chunks.push(currentChunk);

            let trueLineWidth = 0;
            chunks.forEach(ck => {
                let fontStyle = (ck.bold ? "bold " : "") + (ck.italic ? "italic " : "");
                context.font = `${fontStyle}${node.fontSize}px 'Malgun Gothic'`;
                ck.width = context.measureText(ck.text).width;
                trueLineWidth += ck.width;
            });
            return { chunks, trueLineWidth };
        });

        const cache = {
            text: node.text,
            fontSize: node.fontSize,
            width: node.width,
            textAlign: node.textAlign || "center",
            lh,
            totalTextHeight,
            preparedLines
        };
        node._layoutCache = cache;
        return cache;
    }

    function wrapStyledTextAndResize(context, node) {
        const layout = getNodeLayout(context, node);
        if (!layout) return { lines: [], lh: 0, startY: 0 };
        return { lines: layout.preparedLines, lh: layout.lh, startY: node.y - (layout.totalTextHeight / 2) + (layout.lh / 2) };
    }

    function getBestPoints(f, t, ms, me) {
        if (!f || !t) return {f:{x:0,y:0}, t:{x:0,y:0}}; const gp = (n) => ({top:{x:n.x,y:n.y-n.height/2}, bottom:{x:n.x,y:n.y+n.height/2}, left:{x:n.x-n.width/2,y:n.y}, right:{x:n.x+n.width/2,y:n.y}});
        const fp=gp(f), tp=t.width?gp(t):{a:t}; let bf = ms?fp[ms]:null, bt = me?tp[me]:null;
        if(!bf || !bt) { let m=Infinity; const fl=ms?[fp[ms]]:Object.values(fp), tl=me?[tp[me]]:(t.width?Object.values(tp):[t]); fl.forEach(fpx => tl.forEach(tpx => { let d=Math.hypot(fpx.x-tpx.x, fpx.y-tpx.y); if(d<m){m=d; if(!ms)bf=fpx; if(!me)bt=tpx;}})); } return {f:bf, t:bt};
    }
    function distToSegment(p, v, w) { const l2 = (v.x-w.x)**2+(v.y-w.y)**2; if(l2===0) return Math.hypot(p.x-v.x,p.y-v.y); let t = Math.max(0,Math.min(1,((p.x-v.x)*(w.x-v.x)+(p.y-v.y)*(w.y-v.y))/l2)); return Math.hypot(p.x-(v.x+t*(w.x-v.x)),p.y-(v.y+t*(w.y-v.y))); }
    function getHandleAt(n, wx, wy) { const s = (HANDLE_SIZE + 5) / currentScale, l = n.x - n.width/2, r = n.x + n.width/2, t = n.y - n.height/2, b = n.y + n.height/2; if (Math.hypot(wx - l, wy - t) < s) return 'tl'; if (Math.hypot(wx - r, wy - t) < s) return 'tr'; if (Math.hypot(wx - l, wy - b) < s) return 'bl'; if (Math.hypot(wx - r, wy - b) < s) return 'br'; return null; }
    function getNearestSide(n, wp) { const d = [Math.abs(wp.y-(n.y-n.height/2)), Math.abs(wp.y-(n.y+n.height/2)), Math.abs(wp.x-(n.x-n.width/2)), Math.abs(wp.x-(n.x+n.width/2))]; const m = Math.min(...d); return m===d[0]?'top':m===d[1]?'bottom':m===d[2]?'left':'right'; }

    function drawCurvedArrow(ctx, f, t, color, isGhost = false, isSelected = false, ms, me, dir = 0, curveSide = 1) {
        if (!f || !t || f.x === undefined || t.x === undefined || isNaN(f.x) || isNaN(t.x)) return; 
        
        const ps = getBestPoints(f, t, ms, me); 
        const midX = (ps.f.x + ps.t.x) / 2, midY = (ps.f.y + ps.t.y) / 2, 
              dx = ps.t.x - ps.f.x, dy = ps.t.y - ps.f.y, 
              len = Math.hypot(dx, dy) || 1, 
              offset = len * 0.11 * (curveSide || 1), 
              nx = -dy / len, ny = dx / len, 
              cpX = midX + nx * offset, cpY = midY + ny * offset; 

        ctx.save(); 
        ctx.globalAlpha = isSelected ? 1.0 : 0.75; 
        const arrowColor = (isSelected && !isChangingColor) ? "#2196f3" : color;
        ctx.strokeStyle = arrowColor;
        ctx.fillStyle = arrowColor;

        if (isGhost) { 
            ctx.setLineDash([5, 5]); 
            ctx.lineWidth = 1.5 / currentScale; 
            ctx.beginPath(); 
            ctx.moveTo(ps.f.x, ps.f.y); 
            ctx.quadraticCurveTo(cpX, cpY, ps.t.x, ps.t.y); 
            ctx.stroke(); 
        } else { 
            // 캘리그라피형 테이퍼드 리본 곡선: 단 1회의 fill()로 GPU 드로우 콜을 최소화하면서 유려한 굵기 변화 구현
            const steps = Math.min(14, Math.max(8, Math.floor(len / 18)));
            const topPts = [];
            const botPts = [];
            const baseW = (isSelected ? 3.4 : 2.2) / currentScale;

            for (let i = 0; i <= steps; i++) {
                const tParam = i / steps;
                const invT = 1 - tParam;
                const bx = invT * invT * ps.f.x + 2 * invT * tParam * cpX + tParam * tParam * ps.t.x;
                const by = invT * invT * ps.f.y + 2 * invT * tParam * cpY + tParam * tParam * ps.t.y;
                
                const tx = 2 * invT * (cpX - ps.f.x) + 2 * tParam * (ps.t.x - cpX);
                const ty = 2 * invT * (cpY - ps.f.y) + 2 * tParam * (ps.t.y - cpY);
                const tLen = Math.hypot(tx, ty) || 1;
                const px = -ty / tLen;
                const py = tx / tLen;

                // 시작과 끝은 날렵하고(0.55x), 중앙은 도톰하게(1.25x) 흐르는 붓글씨형 두께
                const halfW = (baseW * (0.55 + 0.70 * Math.sin(Math.PI * tParam))) / 2;

                topPts.push({ x: bx + px * halfW, y: by + py * halfW });
                botPts.push({ x: bx - px * halfW, y: by - py * halfW });
            }

            ctx.beginPath();
            ctx.moveTo(topPts[0].x, topPts[0].y);
            for (let i = 1; i <= steps; i++) {
                ctx.lineTo(topPts[i].x, topPts[i].y);
            }
            ctx.lineTo(botPts[steps].x, botPts[steps].y);
            for (let i = steps - 1; i >= 0; i--) {
                ctx.lineTo(botPts[i].x, botPts[i].y);
            }
            ctx.closePath();
            ctx.fill();
        }

        const drawHead = (x, y, angle) => { 
            const h = (isSelected ? 15 : 11) / currentScale; 
            ctx.beginPath(); 
            ctx.moveTo(x, y); 
            ctx.lineTo(x - h * Math.cos(angle - Math.PI / 6.5), y - h * Math.sin(angle - Math.PI / 6.5)); 
            // 뒤쪽 날개가 부드럽게 유선형으로 파여 선과 조화롭게 연결되는 세련된 화살촉
            ctx.quadraticCurveTo(
                x - h * 0.70 * Math.cos(angle), 
                y - h * 0.70 * Math.sin(angle), 
                x - h * Math.cos(angle + Math.PI / 6.5), 
                y - h * Math.sin(angle + Math.PI / 6.5)
            ); 
            ctx.closePath(); 
            ctx.fill(); 
        };

        if (dir === 0 || dir === 2) drawHead(ps.t.x, ps.t.y, Math.atan2(ps.t.y - cpY, ps.t.x - cpX)); 
        if (dir === 1 || dir === 2) drawHead(ps.f.x, ps.f.y, Math.atan2(ps.f.y - cpY, ps.f.x - cpX));
        
        if (isSelected && !isChangingColor) { 
            ctx.globalAlpha = 1.0; 
            ctx.fillStyle = "white"; 
            ctx.strokeStyle = "#2196f3"; 
            ctx.lineWidth = 2 / currentScale; 
            [ps.f, ps.t].forEach(p => { 
                ctx.beginPath(); 
                ctx.arc(p.x, p.y, 5 / currentScale, 0, Math.PI * 2); 
                ctx.fill(); 
                ctx.stroke(); 
            }); 
        } 
        ctx.restore();
    }

    function areaContainsNodes(area, allNodes) { return allNodes.some(n => n !== area && n.type !== "area" && n.x > area.x - area.width/2 && n.x < area.x + area.width/2 && n.y > area.y - area.height/2 && n.y < area.y + area.height/2); }

    function isNodeInView(n, vX1, vY1, vX2, vY2) {
        if (!n) return false;
        const hw = (n.width || 60) / 2, hh = (n.height || 60) / 2;
        return (n.x + hw >= vX1 && n.x - hw <= vX2 && n.y + hh >= vY1 && n.y - hh <= vY2);
    }

    function isEdgeInView(e, vX1, vY1, vX2, vY2) {
        if (!e || !e.from || !e.to) return false;
        const minX = Math.min(e.from.x, e.to.x) - 40, maxX = Math.max(e.from.x, e.to.x) + 40;
        const minY = Math.min(e.from.y, e.to.y) - 40, maxY = Math.max(e.from.y, e.to.y) + 40;
        return (maxX >= vX1 && minX <= vX2 && maxY >= vY1 && minY <= vY2);
    }

    const drawNode = (n) => {
        if (!n) return; const sel = selectedNodes.includes(n); ctx.save();
        if (n.type !== "area") { 
            ctx.shadowColor = "rgba(0,0,0,0.12)"; 
            ctx.shadowBlur = 4; 
            ctx.shadowOffsetX = 2; 
            ctx.shadowOffsetY = 2; 
        }
        
        if (n.type === "marker") { 
            ctx.beginPath(); ctx.arc(n.x, n.y, 15, 0, Math.PI * 2); ctx.fillStyle = COLORS[n.colorIndex || 0]; ctx.fill(); 
            ctx.strokeStyle = sel ? "#2196f3" : "#e0dcc5"; ctx.lineWidth = sel ? 3 : 1.5; ctx.stroke(); 
            ctx.fillStyle = "white"; ctx.font = "bold 13px Arial"; ctx.textAlign = "center"; ctx.textBaseline = "middle"; ctx.fillText(n.value, n.x, n.y); 
        } 
        else if (n.type === "image") {
            let img = imageCache.get(n.data); 
            if (!img) { img = new Image(); img.onload = () => requestRender(); img.src = n.data; imageCache.set(n.data, img); }
            ctx.beginPath(); ctx.roundRect(n.x-n.width/2, n.y-n.height/2, n.width, n.height, 8); ctx.fillStyle = "white"; ctx.fill();
            ctx.save(); ctx.shadowColor = "transparent"; ctx.beginPath(); ctx.roundRect(n.x-n.width/2, n.y-n.height/2, n.width, n.height, 8); ctx.clip(); 
            if (img.complete) ctx.drawImage(img, n.x-n.width/2, n.y-n.height/2, n.width, n.height); 
            ctx.restore();
            ctx.shadowColor = "transparent"; ctx.strokeStyle = sel ? "#2196f3" : "#cccccc"; ctx.lineWidth = 3; ctx.strokeRect(n.x-n.width/2, n.y-n.height/2, n.width, n.height);
        }
        else if (n.type === "area") { 
            ctx.beginPath(); ctx.roundRect(n.x-n.width/2, n.y-n.height/2, n.width, n.height, 5); ctx.fillStyle = `rgba(255, 255, 255, 0.05)`; ctx.fill(); 
            ctx.shadowColor = "transparent"; ctx.setLineDash([2, 3]); ctx.strokeStyle = COLORS[n.colorIndex || 0]; ctx.globalAlpha = sel ? 1.0 : 0.7; ctx.lineWidth = (sel ? 4 : 2) / currentScale; ctx.stroke(); 
        } 
        else { 
            const layout = getNodeLayout(ctx, n);
            let al = n.opacity !== undefined ? n.opacity : 1.0; 
            if (selectedNodes.length > 1 && sel) al *= 0.5;
            
            if (n.type !== "4") { 
                ctx.beginPath(); ctx.roundRect(n.x-n.width/2, n.y-n.height/2, n.width, n.height, 5); 
                ctx.fillStyle = `rgba(255, 255, 255, ${al})`; ctx.fill(); 
                ctx.shadowColor = "transparent"; ctx.strokeStyle = sel ? "#2196f3" : "#5d5b4a"; ctx.lineWidth = (sel ? 2.5 : 1.5) / currentScale; 
                if (n.type === "1") ctx.stroke(); 
                else if (n.type === "2") { ctx.beginPath(); ctx.moveTo(n.x-n.width/2, n.y+n.height/2); ctx.lineTo(n.x+n.width/2, n.y+n.height/2); ctx.stroke(); } 
            } else if (sel) { 
                ctx.beginPath(); ctx.roundRect(n.x-n.width/2, n.y-n.height/2, n.width, n.height, 5); 
                ctx.shadowColor = "transparent"; ctx.strokeStyle = "rgba(33, 150, 243, 0.5)"; ctx.lineWidth = 1.5 / currentScale; ctx.setLineDash([5, 5]); ctx.stroke(); ctx.setLineDash([]); 
            }
            
            ctx.shadowColor = "transparent";
            ctx.globalAlpha = 1.0; 
            const defaultColor = COLORS[n.colorIndex || 0]; 
            ctx.textBaseline = "middle"; 
            
            if (layout) {
                const startY = n.y - (layout.totalTextHeight / 2) + (layout.lh / 2);
                layout.preparedLines.forEach((lineInfo, i) => {
                    let curX = n.x - lineInfo.trueLineWidth / 2; 
                    if (n.textAlign === "left") curX = n.x - n.width / 2 + 15; 
                    if (n.textAlign === "right") curX = n.x + n.width / 2 - 15 - lineInfo.trueLineWidth;
                    ctx.textAlign = "left";
                    lineInfo.chunks.forEach(ck => {
                        let fontStyle = (ck.bold ? "bold " : "") + (ck.italic ? "italic " : ""); 
                        ctx.font = `${fontStyle}${n.fontSize}px 'Malgun Gothic'`; 
                        ctx.fillStyle = ck.color || defaultColor; 
                        ctx.fillText(ck.text, curX, startY + (i * layout.lh));
                        if (ck.underline) { 
                            ctx.beginPath(); 
                            ctx.moveTo(curX, startY + (i * layout.lh) + (n.fontSize * 0.45)); 
                            ctx.lineTo(curX + ck.width, startY + (i * layout.lh) + (n.fontSize * 0.45)); 
                            ctx.strokeStyle = ck.color || defaultColor; 
                            ctx.lineWidth = 1.5; 
                            ctx.stroke(); 
                        } 
                        curX += ck.width;
                    });
                });
            }
        }
        if (sel && !["marker"].includes(n.type)) { 
            ctx.shadowColor = "transparent";
            ctx.fillStyle = "#2196f3"; 
            ctx.globalAlpha = 1.0; 
            const s = 7 / currentScale; 
            [[-1,-1],[1,-1],[-1,1],[1,1]].forEach(c => ctx.fillRect(n.x+(n.width/2)*c[0]-s/2, n.y+(n.height/2)*c[1]-s/2, s, s)); 
        }
        ctx.restore();
    };

    function render() {
        ctx.clearRect(0,0,canvas.width,canvas.height); 
        ctx.fillStyle = bgSettings.color; 
        ctx.fillRect(0,0,canvas.width,canvas.height); 
        ctx.save(); 
        ctx.translate(currentOffset.x, currentOffset.y); 
        ctx.scale(currentScale, currentScale);

        // Viewport bounds in world coordinates with safe margin:
        const margin = 100 / currentScale;
        const vX1 = -currentOffset.x / currentScale - margin;
        const vY1 = -currentOffset.y / currentScale - margin;
        const vX2 = (canvas.width - currentOffset.x) / currentScale + margin;
        const vY2 = (canvas.height - currentOffset.y) / currentScale + margin;

        if (bgSettings.showGrid) { 
            ctx.beginPath(); 
            ctx.strokeStyle = "rgba(0,0,0,0.05)"; 
            ctx.lineWidth = 1 / currentScale; 
            const startX = Math.floor(vX1 / GRID_SIZE) * GRID_SIZE;
            const endX = Math.ceil(vX2 / GRID_SIZE) * GRID_SIZE;
            const startY = Math.floor(vY1 / GRID_SIZE) * GRID_SIZE;
            const endY = Math.ceil(vY2 / GRID_SIZE) * GRID_SIZE;
            for (let x = startX; x <= endX; x += GRID_SIZE) { ctx.moveTo(x, startY); ctx.lineTo(x, endY); } 
            for (let y = startY; y <= endY; y += GRID_SIZE) { ctx.moveTo(startX, y); ctx.lineTo(endX, y); } 
            ctx.stroke(); 
        }

        if (isSelecting || isCreatingArea) { 
            ctx.setLineDash(isCreatingArea ? [2, 2] : [5, 5]); 
            ctx.fillStyle = "rgba(33,150,243,0.1)"; 
            ctx.strokeStyle = "rgba(33,150,243,0.5)"; 
            ctx.fillRect(selectionRect.x1, selectionRect.y1, selectionRect.x2-selectionRect.x1, selectionRect.y2-selectionRect.y1); 
            ctx.strokeRect(selectionRect.x1, selectionRect.y1, selectionRect.x2-selectionRect.x1, selectionRect.y2-selectionRect.y1); 
            ctx.setLineDash([]); 
        }

        const nonAreas = nodes.filter(n => n.type !== "area"), areas = nodes.filter(n => n.type === "area");
        const containedAreas = areas.filter(a => areaContainsNodes(a, nonAreas)), emptyAreas = areas.filter(a => !containedAreas.includes(a));
        
        containedAreas.forEach(n => { if (isNodeInView(n, vX1, vY1, vX2, vY2)) drawNode(n); }); 
        nonAreas.forEach(n => { if (isNodeInView(n, vX1, vY1, vX2, vY2)) drawNode(n); }); 
        emptyAreas.forEach(n => { if (isNodeInView(n, vX1, vY1, vX2, vY2)) drawNode(n); });
        
        edges.forEach(e => { 
            if (e && e.from && e.to && !isNaN(e.from.x) && !isNaN(e.to.x)) { 
                if (isEdgeInView(e, vX1, vY1, vX2, vY2)) {
                    const isEdgeSel = (e === selectedEdge || (selectedNodes.includes(e.from) && selectedNodes.includes(e.to))); 
                    drawCurvedArrow(ctx, e.from, e.to, COLORS[e.colorIndex||0], false, isEdgeSel, e.manualStart, e.manualEnd, e.dirType, e.curveSide); 
                }
            } 
        });
        
        if (activeImageData !== null && activeImageDimensions !== null) { 
            let ghostImg = imageCache.get(activeImageData); 
            let gw = activeImageDimensions.w, gh = activeImageDimensions.h; 
            ctx.save(); ctx.globalAlpha = 0.5; 
            if (ghostImg && ghostImg.complete) ctx.drawImage(ghostImg, currentWorldMouse.x - gw/2, currentWorldMouse.y - gh/2, gw, gh); 
            else { ctx.setLineDash([5,5]); ctx.strokeStyle="#aaa"; ctx.strokeRect(currentWorldMouse.x - gw/2, currentWorldMouse.y - gh/2, gw, gh); } 
            ctx.restore(); 
        }
        if (activeMarkerValue !== null) { 
            ctx.save(); ctx.globalAlpha = 0.4; 
            ctx.beginPath(); ctx.arc(currentWorldMouse.x, currentWorldMouse.y, 15, 0, Math.PI * 2); 
            ctx.fillStyle = COLORS[0]; ctx.fill(); ctx.strokeStyle = "#e0dcc5"; ctx.stroke(); 
            ctx.fillStyle = "white"; ctx.font = "bold 13px Arial"; ctx.textAlign = "center"; ctx.textBaseline = "middle"; 
            ctx.fillText(activeMarkerValue, currentWorldMouse.x, currentWorldMouse.y); 
            ctx.restore(); 
        }
        if (connectingNode) drawCurvedArrow(ctx, connectingNode, currentWorldMouse, "rgba(33,150,243,0.6)", true);
        ctx.restore();
    }

    function animate() { 
        const scaleDiff = Math.abs(targetScale - currentScale);
        const offsetXDiff = Math.abs(targetOffset.x - currentOffset.x);
        const offsetYDiff = Math.abs(targetOffset.y - currentOffset.y);
        const isInterpolating = scaleDiff > 0.0005 || offsetXDiff > 0.1 || offsetYDiff > 0.1;

        if (isInterpolating) {
            currentScale += (targetScale - currentScale) * LERP_FACTOR; 
            currentOffset.x += (targetOffset.x - currentOffset.x) * LERP_FACTOR; 
            currentOffset.y += (targetOffset.y - currentOffset.y) * LERP_FACTOR; 
            zoomInfo.innerText = Math.round(currentScale * 100) + "%"; 
            needsRender = true;
        } else if (currentScale !== targetScale || currentOffset.x !== targetOffset.x || currentOffset.y !== targetOffset.y) {
            currentScale = targetScale;
            currentOffset.x = targetOffset.x;
            currentOffset.y = targetOffset.y;
            zoomInfo.innerText = Math.round(currentScale * 100) + "%";
            needsRender = true;
        }

        if (needsRender) {
            render(); 
            needsRender = false;
        }

        requestAnimationFrame(animate); 
    }
